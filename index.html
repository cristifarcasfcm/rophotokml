async function injectEXIF(blob, datetime){ try{ const dataUrl = await blobToDataURL(blob); const zeroth = {}; const exif = {}; const gps = {}; const pad=n=>String(n).padStart(2,'0'); const dtStr = datetime.getFullYear()+':' + pad(datetime.getMonth()+1)+':' + pad(datetime.getDate())+' ' + pad(datetime.getHours())+':' + pad(datetime.getMinutes())+':' + pad(datetime.getSeconds()); // DATA / ORA zeroth[piexif.ImageIFD.DateTime] = dtStr; exif[piexif.ExifIFD.DateTimeOriginal] = dtStr; exif[piexif.ExifIFD.DateTimeDigitized] = dtStr; // ORIENTARE zeroth[piexif.ImageIFD.Orientation] = 1; // ALTITUDINE if(alt != null){ gps[piexif.GPSIFD.GPSAltitudeRef] = alt < 0 ? 1 : 0; gps[piexif.GPSIFD.GPSAltitude] = [Math.abs(Math.round(alt * 100)),100]; } // COORDONATE GPS if(lat != null && lon != null){ gps[piexif.GPSIFD.GPSLatitudeRef] = lat >= 0 ? "N" : "S"; gps[piexif.GPSIFD.GPSLongitudeRef] = lon >= 0 ? "E" : "W"; gps[piexif.GPSIFD.GPSLatitude] = toDMS(Math.abs(lat)); gps[piexif.GPSIFD.GPSLongitude] = toDMS(Math.abs(lon)); } const exifObj = { "0th": zeroth, "Exif": exif, "GPS": gps }; const exifBytes = piexif.dump(exifObj); const newDataUrl = piexif.insert(exifBytes,dataUrl); return dataURLtoBlob(newDataUrl); }catch(e){ console.error("EXIF ERROR:",e); return blob; } } function toDMS(value){ const deg = Math.floor(value); const minFloat = (value - deg) * 60; const min = Math.floor(minFloat); const sec = (minFloat - min) * 60; return [ [deg,1], [min,1], [Math.round(sec * 100),100] ]; } function blobToDataURL(blob){ return new Promise(resolve=>{ const r = new FileReader(); r.onload = ()=>resolve(r.result); r.readAsDataURL(blob); }); } function dataURLtoBlob(dataurl){ const arr = dataurl.split(','); const mime = arr[0].match(/:(.*?);/)[1]; const bstr = atob(arr[1]); let n = bstr.length; const u8arr = new Uint8Array(n); while(n--){ u8arr[n] = bstr.charCodeAt(n); } return new Blob([u8arr],{type:mime}); }
